# Phase 19: Report Validator

## Objective

Build the report validation system that cross-checks generated reports against source data to ensure accuracy. Validation includes structural checks, numerical verification, and optional AI-assisted narrative review.

## Why This Phase Is Nineteenth

Validation runs after the report is built. It requires the complete pipeline (Builder, Output Adapters) to produce reports to validate, and the Claude CLI client for AI-assisted narrative checking. All upstream dependencies are complete.

from pygramattic_reports.models import (
    Report, TemplateSpec, ValidationCheck, CheckStatus, SectionType,
)


class StructuralValidator:
    """Validates report structure against the template.

    Checks:
    - All required sections from the template are present
    - Data table sections contain data (non-empty headers and rows)
    - Chart sections contain image data
    - No sections are missing titles where required
    - Section ordering matches template
    """

    def validate(self, report: Report, template: TemplateSpec) -> list[ValidationCheck]:
        """Run all structural validation checks.

        Args:
            report: The generated report
            template: The template it was built from

        Returns:
            List of ValidationCheck results
        """
        checks = []
        checks.extend(self._check_section_count(report, template))
        checks.extend(self._check_section_types(report, template))
        checks.extend(self._check_data_tables(report))
        checks.extend(self._check_charts(report))
        checks.extend(self._check_titles(report))
        return checks

    def _check_section_count(self, report, template):
        """Verify the expected number of sections were generated."""
        expected = len(template.sections)
        actual = len(report.sections)
        status = CheckStatus.PASS if actual >= expected - len(report.build_warnings) else CheckStatus.FAIL
        return [ValidationCheck(
            check_name="section_count",
            status=status,
            message=f"Expected ~{expected} sections, got {actual}",
            expected=str(expected),
            actual=str(actual),
        )]

    def _check_data_tables(self, report):
        """Verify data table sections have content."""
        checks = []
        for i, section in enumerate(report.sections):
            if section.section_type == SectionType.DATA_TABLE:
                has_data = (
                    section.table_data is not None
                    and len(section.table_data.get("headers", [])) > 0
                    and len(section.table_data.get("rows", [])) > 0
                )
                checks.append(ValidationCheck(
                    check_name=f"data_table_content_{i}",
                    status=CheckStatus.PASS if has_data else CheckStatus.FAIL,
                    message=f"Data table section {i} {'has' if has_data else 'is missing'} content",
                    section_index=i,
                ))
        return checks

    def _check_charts(self, report):
        """Verify chart sections have image data."""
        checks = []
        for i, section in enumerate(report.sections):
            if section.section_type == SectionType.CHART:
                has_image = section.media_bytes is not None and len(section.media_bytes) > 0
                checks.append(ValidationCheck(
                    check_name=f"chart_image_{i}",
                    status=CheckStatus.PASS if has_image else CheckStatus.WARN,
                    message=f"Chart section {i} {'has' if has_image else 'is missing'} image data",
                    section_index=i,
                    severity="warning" if not has_image else "info",
                ))
        return checks

### Task 19.2: Implement the Numerical Validator

**File to create:** `/Volumes/SecondDrive/code2/pygramattic-reports/src/pygramattic_reports/validator/numerical.py`

**Description:** Validates numerical claims in the report against source datasets. Uses the `NumberClaim` list generated by the Builder (not AI -- pure deterministic comparison).

```python
from pygramattic_reports.models import (
    Report, NumberClaim, Dataset, ValidationCheck, CheckStatus,
)
from pygramattic_reports.processors import aggregate


class NumericalValidator:
    """Validates numerical claims in reports against source data.

    Uses the NumberClaim manifest from the Builder to deterministically
    verify every number in the report. This does NOT use AI -- it is
    pure programmatic comparison.

    Checks:
    - Each NumberClaim's value matches the source dataset
    - Computed values (sums, averages) are correct
    - No NaN or infinite values in claims
    """

    def __init__(self, tolerance: float = 0.01):
        """
        Args:
            tolerance: Relative tolerance for float comparison (1% default)
        """
        self.tolerance = tolerance

    def validate(
        self,
        report: Report,
        datasets: dict[str, Dataset],
    ) -> list[ValidationCheck]:
        """Validate all numerical claims.

        Args:
            report: The report to validate
            datasets: Map of dataset_id -> Dataset for source verification

        Returns:
            List of ValidationCheck results
        """
        checks = []
        for claim in report.number_claims:
            check = self._verify_claim(claim, datasets)
            checks.append(check)
        return checks

    def _verify_claim(self, claim: NumberClaim, datasets: dict[str, Dataset]) -> ValidationCheck:
        """Verify a single numerical claim against source data."""
        dataset = datasets.get(claim.source_dataset_id)
        if dataset is None:
            return ValidationCheck(
                check_name=f"number_claim_{claim.description}",
                status=CheckStatus.SKIP,
                message=f"Source dataset {claim.source_dataset_id} not found",
                section_index=claim.section_index,
            )

        try:
            expected = self._compute_expected(claim, dataset)
            if self._values_match(claim.value, expected):
                return ValidationCheck(
                    check_name=f"number_claim_{claim.description}",
                    status=CheckStatus.PASS,
                    message=f"{claim.description}: {claim.formatted_value} matches source",
                    expected=str(expected),
                    actual=str(claim.value),
                    section_index=claim.section_index,
                )
            else:
                return ValidationCheck(
                    check_name=f"number_claim_{claim.description}",
                    status=CheckStatus.FAIL,
                    message=f"{claim.description}: expected {expected}, got {claim.value}",
                    expected=str(expected),
                    actual=str(claim.value),
                    section_index=claim.section_index,
                )
        except Exception as e:
            return ValidationCheck(
                check_name=f"number_claim_{claim.description}",
                status=CheckStatus.WARN,
                message=f"Could not verify: {e}",
                section_index=claim.section_index,
                severity="warning",
            )

    def _compute_expected(self, claim: NumberClaim, dataset: Dataset) -> float:
        """Re-compute the expected value from the source dataset."""
        if claim.computation.startswith("raw["):
            # Direct value from a specific row
            row_idx = int(claim.computation.split("[")[1].rstrip("]"))
            return float(dataset.dataframe[claim.source_column].iloc[row_idx])
        elif claim.computation in ("sum", "avg", "min", "max", "count"):
            return aggregate(dataset, claim.source_column, claim.computation)
        else:
            raise ValueError(f"Unknown computation: {claim.computation}")

    def _values_match(self, actual: float, expected: float) -> bool:
        """Compare two float values with tolerance."""
        if expected == 0:
            return abs(actual) < self.tolerance
        return abs((actual - expected) / expected) < self.tolerance
```

### Task 19.3: Implement the Narrative Validator (AI-Assisted)

**File to create:** `/Volumes/SecondDrive/code2/pygramattic-reports/src/pygramattic_reports/validator/narrative.py`

**Description:** Uses Claude CLI to check narrative sections for factual alignment with data. This is the only validation step that uses AI.

```python
import json
from pygramattic_reports.models import Report, Dataset, SectionType, ValidationCheck, CheckStatus
from pygramattic_reports.ai import ClaudeClient
from pygramattic_reports.ai.prompts import VALIDATION_PROMPT
from pygramattic_reports.exceptions import AIError
from pygramattic_reports.logging import get_logger

logger = get_logger("validator.narrative")


class NarrativeValidator:
    """AI-assisted validation of narrative report sections.

    Uses Claude CLI to check whether narrative text accurately
    reflects the underlying data. This is optional -- if AI is
    unavailable, narrative validation is skipped.
    """

    def __init__(self, client: ClaudeClient | None):
        self.client = client

    def validate(
        self,
        report: Report,
        datasets: dict[str, Dataset],
    ) -> list[ValidationCheck]:
        """Validate narrative sections using AI.

        Skips validation if Claude CLI is unavailable.
        """
        if self.client is None or not self.client.is_available():
            return [ValidationCheck(
                check_name="narrative_validation",
                status=CheckStatus.SKIP,
                message="Narrative validation skipped: AI unavailable",
            )]

        checks = []
        for i, section in enumerate(report.sections):
            if section.section_type in (SectionType.SUMMARY, SectionType.NARRATIVE):
                check = self._validate_section(section, i, datasets)
                checks.append(check)
        return checks

    def _validate_section(self, section, index, datasets):
        """Validate a single narrative section."""
        # Build data context from all datasets
        data_summary = "\n".join(
            f"{name}: {ds.dataframe.describe().to_string()}"
            for name, ds in datasets.items()
        )

        prompt = VALIDATION_PROMPT.format(
            report_text=section.content or "",
            source_data=data_summary,
        )

        try:
            response = self.client.generate(prompt=prompt)
            result = json.loads(response)
            valid = result.get("valid", True)
            issues = result.get("issues", [])

            if valid:
                return ValidationCheck(
                    check_name=f"narrative_accuracy_{index}",
                    status=CheckStatus.PASS,
                    message=f"Narrative section {index} is consistent with data",
                    section_index=index,
                )
            else:
                return ValidationCheck(
                    check_name=f"narrative_accuracy_{index}",
                    status=CheckStatus.WARN,
                    message=f"Narrative issues: {'; '.join(issues)}",
                    section_index=index,
                    severity="warning",
                )
        except (AIError, json.JSONDecodeError) as e:
            return ValidationCheck(
                check_name=f"narrative_accuracy_{index}",
                status=CheckStatus.SKIP,
                message=f"Narrative validation failed: {e}",
                section_index=index,
            )
```

### Task 19.4: Implement the Validation Orchestrator

**File to create:** `/Volumes/SecondDrive/code2/pygramattic-reports/src/pygramattic_reports/validator/validator.py`

**Description:** Orchestrates all three validation layers and produces the final `ValidationResult`.

```python
from pygramattic_reports.models import (
    Report, TemplateSpec, Dataset, ValidationResult, CheckStatus, now_utc,
)
from pygramattic_reports.ai import ClaudeClient
from .structural import StructuralValidator
from .numerical import NumericalValidator
from .narrative import NarrativeValidator


class ReportValidator:
    """Full report validation orchestrator.

    Runs three layers of validation:
    1. Structural: template compliance, section completeness
    2. Numerical: data accuracy, computation correctness
    3. Narrative: AI-assisted factual alignment (optional)

    Usage:
        validator = ReportValidator(claude_client=client)
        result = validator.validate(report, template, datasets)
        if result.overall_status == CheckStatus.FAIL:
            print("Validation failed!")
            for check in result.checks:
                if check.status == CheckStatus.FAIL:
                    print(f"  - {check.message}")
    """

    def __init__(self, claude_client: ClaudeClient | None = None):
        self.structural = StructuralValidator()
        self.numerical = NumericalValidator()
        self.narrative = NarrativeValidator(claude_client)

    def validate(
        self,
        report: Report,
        template: TemplateSpec,
        datasets: dict[str, Dataset],
    ) -> ValidationResult:
        """Run all validation checks and produce a ValidationResult."""
        all_checks = []

        # Layer 1: Structural
        all_checks.extend(self.structural.validate(report, template))

        # Layer 2: Numerical
        dataset_map = {ds.id: ds for ds in datasets.values()}
        all_checks.extend(self.numerical.validate(report, dataset_map))

        # Layer 3: Narrative (optional, may skip)
        all_checks.extend(self.narrative.validate(report, datasets))

        # Compute overall status
        passed = sum(1 for c in all_checks if c.status == CheckStatus.PASS)
        failed = sum(1 for c in all_checks if c.status == CheckStatus.FAIL)
        warnings = sum(1 for c in all_checks if c.status == CheckStatus.WARN)

        if failed > 0:
            overall = CheckStatus.FAIL
        elif warnings > 0:
            overall = CheckStatus.WARN
        else:
            overall = CheckStatus.PASS

        return ValidationResult(
            report_id=report.id,
            overall_status=overall,
            checks=all_checks,
            total_checks=len(all_checks),
            passed=passed,
            failed=failed,
            warnings=warnings,
            validated_at=now_utc(),
        )
```
